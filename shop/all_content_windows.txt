from django.db import models
from django.contrib.auth.models import User


class Product(models.Model):
    name = models.CharField(max_length=50)
    count = models.SmallIntegerField(default=0)
    discrp = models.CharField(max_length=100)
    price = models.PositiveIntegerField(default=0)
    img  = models.ImageField()
    def __str__(self):
        return self.name
    class Meta:
        verbose_name="Товар"


class Order(models.Model):
    user = models.ForeignKey(User, on_delete=models.SET_NULL, null=True, blank=True)
    full_name = models.CharField(max_length=255, verbose_name="ПІБ")
    phone = models.CharField(max_length=20, verbose_name="Телефон")
    address = models.TextField(verbose_name="Адреса доставки")
    created_at = models.DateTimeField(auto_now_add=True, verbose_name="Дата створення")

    def __str__(self):
        return f"Замовлення #{self.id} - {self.full_name}"

    

class OrderItem(models.Model):
    order = models.ForeignKey(Order, related_name='items', on_delete=models.CASCADE)
    product = models.ForeignKey('Product', on_delete=models.CASCADE) # Замініть Product на назву вашої моделі товарів
    quantity = models.PositiveIntegerField(default=1)
    price = models.DecimalField(max_length=10, decimal_places=2, max_digits=10)

    def __str__(self):
        return f"{self.product.name} x {self.quantity}"

        // 1. Відображення товарів із localStorage
        function renderCart() {
            const cart = JSON.parse(localStorage.getItem('petcare_cart')) || [];
            const cartContainer = document.getElementById('cart-items');
            const totalPriceElement = document.getElementById('total-price');

            if (!cartContainer || !totalPriceElement) return;

            cartContainer.innerHTML = '';
            let total = 0;

            if (cart.length === 0) {
                cartContainer.innerHTML = '<p style="text-align:center; padding: 20px 0;">Ваш кошик порожній</p>';
                totalPriceElement.innerText = '0 грн';
                document.getElementById('checkout-form').style.display = 'none'; // Ховаємо форму, якщо кошик порожній
                return;
            } else {
                document.getElementById('checkout-form').style.display = 'block';
            }

            cart.forEach(item => {
                const itemTotal = item.price * item.quantity;
                total += itemTotal;

                const itemElement = document.createElement('div');
                itemElement.className = 'cart-item';
                itemElement.innerHTML = `
                    <div class="cart-item-info">
                        <h4 style="margin: 0 0 5px 0;">${item.name}</h4>
                        <p style="margin: 0; color: #6b7280;">Ціна: ${item.price} грн</p>
                    </div>
                    
                    <div class="quantity-controls">
                        <button type="button" onclick="updateCartQuantity('${item.id}', -1)">-</button>
                        <span>${item.quantity}</span>
                        <button type="button" onclick="updateCartQuantity('${item.id}', 1)">+</button>
                    </div>

                    <div class="cart-item-total" style="display: flex; align-items: center; gap: 10px;">
                        <strong>${itemTotal} грн</strong>
                        <button type="button" onclick="removeFromCart('${item.id}')" class="delete-btn">Видалити</button>
                    </div>
                `;
                cartContainer.appendChild(itemElement);
            });

            totalPriceElement.innerText = total + ' грн';
        }

        // 2. Зміна кількості товару
        function updateCartQuantity(id, step) {
            let cart = JSON.parse(localStorage.getItem('petcare_cart')) || [];
            const item = cart.find(i => String(i.id) === String(id));

            if (item) {
                item.quantity += step;
                if (item.quantity <= 0) {
                    cart = cart.filter(i => String(i.id) !== String(id));
                }
                localStorage.setItem('petcare_cart', JSON.stringify(cart));
                renderCart();
            }
        }

        // 3. Видалення товару з кошика
        function removeFromCart(id) {
            let cart = JSON.parse(localStorage.getItem('petcare_cart')) || [];
            cart = cart.filter(item => String(item.id) !== String(id));
            localStorage.setItem('petcare_cart', JSON.stringify(cart));
            renderCart();
        }

        // 4. Відправка замовлення в Django через fetch
        function submitOrder(event) {
            event.preventDefault(); // Зупиняємо стандартне перезавантаження сторінки

            const cart = JSON.parse(localStorage.getItem('petcare_cart')) || [];
            if (cart.length === 0) {
                alert("Ваш кошик порожній!");
                return;
            }

            // Збираємо дані з форми та кошика в один об'єкт
            const orderData = {
                full_name: document.getElementById('full_name').value,
                phone: document.getElementById('phone').value,
                address: document.getElementById('address').value,
                items: cart.map(item => ({ id: item.id, quantity: item.quantity }))
            };

            // Отримуємо CSRF-токен для безпечного запиту в Django
            const csrfToken = document.querySelector('[name=csrfmiddlewaretoken]').value;

            // Відправляємо дані на сервер
            fetch('/create-order/', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                    'X-CSRFToken': csrfToken
                },
                body: JSON.stringify(orderData)
            })
            .then(response => response.json())
            .then(data => {
                if (data.status === 'success') {
                    alert('Дякуємо! Ваше замовлення успішно оформлено.');
                    localStorage.removeItem('petcare_cart'); // Очищаємо кошик після успішної покупки
                    window.location.href = '/'; // Перенаправляємо на головну сторінку
                } else {
                    alert('Помилка: ' + data.message);
                }
            })
            .catch(error => {
                console.error('Error:', error);
                alert('Сталася помилка при відправці замовлення.');
            });
        }

        document.addEventListener('DOMContentLoaded', renderCart);
    </script>

            // Функція оновлення лічильника товарів у шапці сайту
            function updateCartCountHeader() {
                const cart = JSON.parse(localStorage.getItem('petcare_cart')) || [];
                const totalCount = cart.reduce((sum, item) => sum + item.quantity, 0);
                const countElement = document.getElementById('cart-count');
                if (countElement) {
                    countElement.innerText = totalCount;
                }
            }

            // 1. Функція зміни кількості (+ / -)
            function changeQuantity(button, step) {
                const card = button.closest('.product-card');
                const qtyElement = card.querySelector('.qty-number');
                let currentQty = parseInt(qtyElement.innerText) || 1;

                if (step === 1) {
                    const maxStock = parseInt(button.getAttribute('data-max')) || 999;
                    if (currentQty < maxStock) {
                        currentQty += 1;
                    } else {
                        alert("На складі немає більше цього товару!");
                    }
                } else if (step === -1 && currentQty > 1) {
                    currentQty -= 1;
                }

                qtyElement.innerText = currentQty;
            }

            // 2. Функція додавання в кошик
            function addToCart(button) {
                const card = button.closest('.product-card');

                const id = button.getAttribute('data-id');
                const name = button.getAttribute('data-name');
                const price = parseFloat(button.getAttribute('data-price')) || 0;

                const qtyElement = card.querySelector('.qty-number');
                const selectedQuantity = parseInt(qtyElement ? qtyElement.innerText : 1) || 1;

                if (!id) {
                    alert("Помилка: Не знайдено ID товару у кнопці!");
                    return;
                }

                let cart = [];
                try {
                    cart = JSON.parse(localStorage.getItem('petcare_cart')) || [];
                } catch (e) {
                    cart = [];
                }

                const existingIndex = cart.findIndex(item => String(item.id) === String(id));

                if (existingIndex !== -1) {
                    cart[existingIndex].quantity += selectedQuantity;
                } else {
                    cart.push({
                        id: String(id),
                        name: name,
                        price: price,
                        quantity: selectedQuantity
                    });
                }

                // Зберігаємо в пам'ять браузера
                localStorage.setItem('petcare_cart', JSON.stringify(cart));

                // Скидаємо лічильник на картці товару назад на 1
                if (qtyElement) qtyElement.innerText = 1;

                // Оновлюємо цифру в шапці сайту
                updateCartCountHeader();

            }

            // Оновлюємо лічильник у шапці при завантаженні сторінки
            document.addEventListener('DOMContentLoaded', updateCartCountHeader);

def create_order(request):
    """Прийом JSON-даних замовлення від JS"""
    if request.method == 'POST':
        try:
            data = json.loads(request.body)
            
            full_name = data.get('full_name')
            phone = data.get('phone')
            address = data.get('address')
            items = data.get('items', [])

            if not items:
                return JsonResponse({'status': 'error', 'message': 'Кошик порожній'}, status=400)

            # 1. Створюємо основне замовлення
            order = Order.objects.create(
                user=request.user if request.user.is_authenticated else None,
                full_name=full_name,
                phone=phone,
                address=address
            )

            # 2. Перебираємо товари із кошика
            for item in items:
                product = Product.objects.get(id=item['id'])
                qty = int(item['quantity'])

                # Створюємо елемент замовлення
                OrderItem.objects.create(
                    order=order,
                    product=product,
                    quantity=qty,
                    price=product.price
                )

                # Зменшуємо залишок на складі (якщо є поле count)
                if hasattr(product, 'count') and product.count >= qty:
                    product.count -= qty
                    product.save()

            return JsonResponse({'status': 'success', 'message': 'Замовлення створено!'})

        except Exception as e:
            return JsonResponse({'status': 'error', 'message': str(e)}, status=500)

    return JsonResponse({'status': 'error', 'message': 'Невідомий метод'}, status=405)
